cmake_minimum_required(VERSION 3.16)
project(EarlyWarningGrid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(early_warning_grid src/main.cpp)
if(MSVC)
  target_compile_options(early_warning_grid PRIVATE /W4)
else()
  target_compile_options(early_warning_grid PRIVATE -Wall -Wextra -pedantic)
endif()
